// Grain Growth Simulation
// Written in JavaScript with p5.js library

let grid = [];
let gridSize = 100; // Number of grains in each direction
let cellSize;
let grains = [];
let time = 0;
let temperature = 0.5; // Temperature (affects growth rate)
let isPaused = false;
let growthRate = 0.01;
let simulationType = 'voronoi'; // 'voronoi' or 'cellular'

// Different colors for grains
const colors = [
  [255, 100, 100], [100, 255, 100], [100, 100, 255],
  [255, 255, 100], [255, 100, 255], [100, 255, 255],
  [200, 150, 100], [150, 200, 100], [100, 150, 200],
  [200, 100, 150], [255, 200, 100], [200, 255, 100]
];

function setup() {
  createCanvas(600, 600);
  cellSize = width / gridSize;
  
  // Initialize grid
  initializeGrid();
  
  // Create control panel
  createControlPanel();
  
  // Start simulation
  initializeGrains();
}

function draw() {
  background(30);
  
  if (!isPaused) {
    updateGrains();
    time++;
  }
  
  drawGrains();
  drawInfo();
}

// Initialize grid
function initializeGrid() {
  grid = [];
  for (let i = 0; i < gridSize; i++) {
    grid[i] = [];
    for (let j = 0; j < gridSize; j++) {
      grid[i][j] = -1; // -1 means no grain
    }
  }
}

// Initialize initial grains
function initializeGrains() {
  grains = [];
  
  if (simulationType === 'voronoi') {
    // Create random grains using Voronoi diagram
    for (let i = 0; i < 50; i++) {
      grains.push({
        x: random(width),
        y: random(height),
        color: colors[i % colors.length],
        size: random(2, 10),
        growthRate: random(0.005, 0.02),
        id: i
      });
    }
  } else {
    // Create grains at random positions
    for (let i = 0; i < 30; i++) {
      grains.push({
        x: floor(random(gridSize)),
        y: floor(random(gridSize)),
        color: colors[i % colors.length],
        size: 1,
        growthRate: random(0.005, 0.02) * temperature,
        id: i
      });
      
      // Set grain in grid
      grid[grains[i].x][grains[i].y] = i;
    }
  }
}

// Update grain growth
function updateGrains() {
  if (simulationType === 'voronoi') {
    // Update grain growth using simplified Voronoi method
    for (let grain of grains) {
      grain.size += grain.growthRate * temperature;
      
      // Reduce growth rate as size increases (simulating Burke's Law)
      grain.growthRate *= 0.999;
      
      // Randomly add new grain
      if (random() < 0.001 * temperature && grains.length < 100) {
        grains.push({
          x: random(width),
          y: random(height),
          color: colors[grains.length % colors.length],
          size: random(1, 3),
          growthRate: random(0.005, 0.01) * temperature,
          id: grains.length
        });
      }
    }
  } else {
    // Update grain growth using cellular propagation method
    let newGrid = [];
    for (let i = 0; i < gridSize; i++) {
      newGrid[i] = [];
      for (let j = 0; j < gridSize; j++) {
        newGrid[i][j] = grid[i][j];
      }
    }
    
    // Update each cell
    for (let i = 1; i < gridSize - 1; i++) {
      for (let j = 1; j < gridSize - 1; j++) {
        // If cell is empty, neighboring grains try to expand into it
        if (grid[i][j] === -1) {
          let neighbors = [];
          
          // Collect cell neighbors
          for (let di = -1; di <= 1; di++) {
            for (let dj = -1; dj <= 1; dj++) {
              if (di === 0 && dj === 0) continue;
              let neighborId = grid[i + di][j + dj];
              if (neighborId !== -1) {
                neighbors.push(neighborId);
              }
            }
          }
          
          // If there are neighbors, choose one randomly
          if (neighbors.length > 0) {
            let selectedGrain = random(neighbors);
            
            // Probability depends on temperature
            if (random() < growthRate * temperature) {
              newGrid[i][j] = selectedGrain;
              
              // Update grain size
              grains[selectedGrain].size += 0.01;
            }
          }
        }
      }
    }
    
    grid = newGrid;
  }
}

// Draw grains
function drawGrains() {
  noStroke();
  
  if (simulationType === 'voronoi') {
    // Draw Voronoi diagram
    for (let x = 0; x < width; x += 2) {
      for (let y = 0; y < height; y += 2) {
        let closestGrain = null;
        let minDist = Infinity;
        
        // Find closest grain
        for (let grain of grains) {
          let d = dist(x, y, grain.x, grain.y) / grain.size;
          if (d < minDist) {
            minDist = d;
            closestGrain = grain;
          }
        }
        
        if (closestGrain) {
          fill(closestGrain.color[0], closestGrain.color[1], closestGrain.color[2], 200);
          rect(x, y, 2, 2);
        }
      }
    }
    
    // Draw grain centers
    for (let grain of grains) {
      fill(255);
      ellipse(grain.x, grain.y, 6, 6);
      fill(grain.color[0], grain.color[1], grain.color[2]);
      ellipse(grain.x, grain.y, 4, 4);
    }
  } else {
    // Draw cellular grid
    for (let i = 0; i < gridSize; i++) {
      for (let j = 0; j < gridSize; j++) {
        if (grid[i][j] !== -1) {
          let grainId = grid[i][j];
          let grain = grains[grainId];
          fill(grain.color[0], grain.color[1], grain.color[2], 220);
          rect(i * cellSize, j * cellSize, cellSize, cellSize);
        }
      }
    }
  }
}

// Draw simulation information
function drawInfo() {
  fill(255);
  stroke(0);
  strokeWeight(1);
  rect(10, 10, 250, 130);
  
  fill(0);
  noStroke();
  textSize(14);
  textAlign(LEFT);
  text("Grain Growth Simulation", 20, 30);
  textSize(12);
  text(`Time: ${time}`, 20, 55);
  text(`Temperature: ${nf(temperature, 1, 2)}`, 20, 75);
  text(`Number of grains: ${grains.length}`, 20, 95);
  text(`Simulation type: ${simulationType === 'voronoi' ? 'Voronoi' : 'Cellular'}`, 20, 115);
  
  // Show status
  if (isPaused) {
    fill(255, 50, 50);
    text("Simulation paused - Press SPACE to resume", 20, 135);
  } else {
    fill(50, 255, 50);
    text("Simulation running - Press SPACE to pause", 20, 135);
  }
}

// Create control panel
function createControlPanel() {
  // Reset button
  let resetButton = createButton('Reset Simulation');
  resetButton.position(width - 150, 20);
  resetButton.mousePressed(resetSimulation);
  
  // Toggle simulation type button
  let toggleButton = createButton('Toggle Simulation Type');
  toggleButton.position(width - 150, 50);
  toggleButton.mousePressed(toggleSimulationType);
  
  // Temperature control slider
  let tempLabel = createP('Temperature:');
  tempLabel.position(width - 150, 80);
  tempLabel.style('color', 'white');
  tempLabel.style('font-size', '12px');
  
  let tempSlider = createSlider(0.1, 1.0, temperature, 0.05);
  tempSlider.position(width - 150, 120);
  tempSlider.input(() => {
    temperature = tempSlider.value();
  });
  
  // Growth rate control slider
  let growthLabel = createP('Growth Rate:');
  growthLabel.position(width - 150, 140);
  growthLabel.style('color', 'white');
  growthLabel.style('font-size', '12px');
  
  let growthSlider = createSlider(0.001, 0.05, growthRate, 0.001);
  growthSlider.position(width - 150, 180);
  growthSlider.input(() => {
    growthRate = growthSlider.value();
  });
}

// Reset simulation
function resetSimulation() {
  time = 0;
  initializeGrid();
  initializeGrains();
}

// Toggle simulation type
function toggleSimulationType() {
  simulationType = simulationType === 'voronoi' ? 'cellular' : 'voronoi';
  resetSimulation();
}

// Control simulation with keyboard
function keyPressed() {
  if (key === ' ') {
    isPaused = !isPaused;
  }
  
  // Increase/decrease temperature
  if (key === 'ArrowUp') {
    temperature = min(1.0, temperature + 0.05);
  } else if (key === 'ArrowDown') {
    temperature = max(0.1, temperature - 0.05);
  }
  
  // Reset simulation
  if (key === 'r' || key === 'R') {
    resetSimulation();
  }
  
  // Toggle simulation type
  if (key === 't' || key === 'T') {
    toggleSimulationType();
  }
}
